#### Platform 对象

```javascript
export const Platform = (() => {
  if (typeof navigator !== 'undefined' && navigator.product === 'ReactNative') {
    return eval('require("react-native").Platform');  // 移动端：原生Platform
  }
  return { OS: 'web', Version: undefined, select: (obj) => obj.web };  // PC端：兼容对象
})();
```

#### ModelPathAdapter 对象

```javascript
export const ModelPathAdapter = {
  getModelBasePath()        // 获取模型基础路径
  getModelPath()            // 获取完整模型路径
  getExecutionProviders()   // 获取推荐的执行提供者
  detectEnvironment()       // 检测当前环境
  loadOnnxRuntime()        // 加载ONNX Runtime
};
```

**平台路径映射**：
- Android: `file:///android_asset/models`
- iOS: `${MainBundlePath}/models`
- PC开发: `http://localhost:3000/models`
- PC生产: `./models`

#### CanvasAdapter 对象

async detectAvailableProviders() {
  // 🆕 使用适配器

  return recommended.filter(p => ortSupported.includes(p));
}
